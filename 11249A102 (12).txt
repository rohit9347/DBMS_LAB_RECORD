SQL> -- Step 5: Function to Find Grade
SQL> CREATE OR REPLACE FUNCTION FIND_GRADE
  2  (
  3   P_MARKS NUMBER
  4  )
  5  RETURN VARCHAR2
  6  IS
  7   V_GRADE VARCHAR2(2);
  8  BEGIN
  9   IF P_MARKS >= 90 THEN
 10    V_GRADE := 'O';
 11   ELSIF P_MARKS >= 80 THEN
 12    V_GRADE := 'A+';
 13   ELSIF P_MARKS >= 70 THEN
 14    V_GRADE := 'A';
 15   ELSIF P_MARKS >= 60 THEN

 19   END IF;

 21  END;
